output
  [episode()]

// ---------- HELPERS ----------

d(n) =>
  return Math.floor(Math.random() * n) + 1;

num(id, fallback) =>
  let el = document.getElementById(id);
  let v = el ? parseInt(el.value) : NaN;
  return isNaN(v) ? fallback : v;

sec(title, body) =>
  return '<section class="sec"><h2>' + title + '</h2>' + body + '</section>';

// ---------- EPISODE ----------

episode() =>
  let renown = num("renownInput", 0);
  let lvl = num("levelInput", 1);
  let cap = Math.max(0, Math.min(renown, lvl));
  window.sejrTier = cap <= 3 ? 1 : (cap <= 7 ? 2 : 3);
  window.sejrRival = false;
  let sel = document.getElementById("dirInput");
  let dir = sel ? sel.value : "Roll";
  let dirNote = "";
  if (dir === "Roll") { let r = d(6); dir = r <= 3 ? "City" : "Arch"; dirNote = " (rolled " + r + ")"; }
  let danger = ["unsafe", "risky", "deadly"][window.sejrTier - 1];
  let head = "";
  let rooms = 3;
  if (dir === "City") {
    let r = d(6);
    let t = r + cap;
    head += "<p><b>City.</b> Stage " + r + " + " + cap + " = " + t + ": <b>" + stage_text(t) + "</b></p>";
    head += "<p><b>Cost:</b> Neglect ticks on " + neglect_place.evaluateItem + ".</p>";
    rooms = d(2) === 1 ? 3 : 4;
    if (t >= 13) { head += "<p>The job leads through an arch, to " + far_side.evaluateItem + ".</p>"; }
  } else {
    let size = site_size.evaluateItem;
    let type = site_type.evaluateItem;
    head += "<p><b>Arch.</b> Patron: " + patron.evaluateItem + ".</p>";
    head += "<p><b>Debt:</b> " + debt.evaluateItem + ".</p>";
    head += "<p><b>Far side:</b> " + far_side.evaluateItem + ". " + size + " " + type.toLowerCase() + ", " + danger + ".</p>";
    rooms = size === "Small" ? 3 : (size === "Medium" ? 4 : 5);
  }
  let hook = detail1.evaluateItem + " " + detail2.evaluateItem.toLowerCase();
  let h = sec("Direction" + dirNote, head);
  h += sec("Hook", "<p class='hook'>" + hook + ", but " + but_table.evaluateItem + ".</p>");
  let map = "<ol>";
  for (let i = 0; i < rooms; i++) { map += "<li>" + room(cap) + "</li>"; }
  map += "</ol>";
  h += sec("Map, " + rooms + " areas", map);
  let sc = "<ol class='scenes'>";
  for (let s = 1; s <= 4; s++) { let r = d(6) + s; sc += "<li><span class='roll'>" + r + "</span> " + butplus(r) + "</li>"; }
  sc += "</ol>";
  h += sec("But+ at each scene change", sc);
  h += sec("Rumor for the thread board", "<p>" + rumor.evaluateItem + ".</p>");
  h += sec("Returning hero", "<p>" + beard.evaluateItem + ".</p>");
  h += "<p class='close'>Close: one thread closes, one opens. Renown +1, capped at average level. Absent players tick an arcana ritual. Neglect advances for skipped places.</p>";
  return h;

stage_text(t) =>
  let s = ["", "Market Square festival, comedy and heist", "Villas of the Second Rise, noir and intrigue", "Public Baths and the Falls of the Three Gods, rumor and romance", "Cliffs of the Dead, ghost story", "Theater of Whispers, masks and secrets", "Arch of Kings, arrivals and the weird"];
  if (t <= 6) return s[t];
  if (t <= 8) return "the Lower Reaches, crime and the Black Hand";
  if (t <= 10) return "the Lost Cisterns, flood horror";
  if (t <= 12) return "Sunken Revvia, dread";
  return "through an arch, genre set by the key";

room(cap) =>
  let r = Math.min(10, d(10) + Math.floor(cap / 3));
  let lists = [room_1, room_2, room_3, room_4, room_5, room_6, room_7, room_8, room_9, room_10];
  return lists[r - 1].evaluateItem;

butplus(r) =>
  if (r <= 2) return "<b>Foreshadow:</b> " + foreshadow.evaluateItem + ".";
  if (r === 3) return "<b>Faction:</b> " + faction_encounter();
  if (r === 4) return "<b>Monster:</b> " + (d(2) === 1 ? solo_adj.evaluateItem + " " + solo_noun.evaluateItem.toLowerCase() : mob_adj.evaluateItem + " " + mob_noun.evaluateItem.toLowerCase()) + ", native to the Stage.";
  if (r === 5) return "<b>Twist:</b> " + but_table.evaluateItem + ".";
  if (r === 6) return "<b>Dusk:</b> light, clock and exits all tighten.";
  return "<b>Dark:</b> the scenario turns, and whatever was coming arrives.";

faction_encounter() =>
  let f = faction.evaluateItem;
  if (f === "Local band") { f = "a local band of " + mob_adj.evaluateItem.toLowerCase() + " " + band_kind.evaluateItem; }
  return f + ", who want " + faction_want.evaluateItem + ". Add to the tracker.";

hero() =>
  let cls = hero_class.evaluateItem;
  let keys = ["Human", "Elf", "Dwarf", "Halfling", "Half-orc", "Goblin"];
  let nm = npc_name[keys[d(6) - 1]].evaluateItem + ' "' + npc_nickname.evaluateItem + '"';
  let ev = (label, list) => "<p><b>" + label + ":</b> " + list.evaluateItem + " <span class='rel'>Tie: " + relationship.evaluateItem + ", " + npc_name[keys[d(6) - 1]].evaluateItem + ".</span></p>";
  let arc = d(6) <= 4 ? "the " + cls + " class arcanum" : other_arcanum.evaluateItem;
  let b = "<p class='hook'>" + nm + ", " + hero_ancestry.evaluateItem + " " + cls + "</p>";
  let key = { Berserker:["STR","DEX"], Cheat:["DEX","INT"], Commander:["STR","INT"], Hunter:["DEX","WIL"], Mage:["INT","WIL"], Oathsworn:["STR","WIL"], Shadowmancer:["INT","DEX"], Shepherd:["WIL","STR"], Songweaver:["WIL","INT"], Stormshifter:["WIL","DEX"], Zephyr:["DEX","STR"] }[cls];
  let t1 = trait.evaluateItem;
  let t2 = t1;
  let g = 0;
  while (t2 === t1 && g < 30) { t2 = trait.evaluateItem; g++; }
  b += "<p><b>Stats:</b> " + stat_array.evaluateItem + ", highest in " + key[0] + " and " + key[1] + ".</p>";
  b += "<p><b>Gifted:</b> " + key[0] + ", " + stat_word[key[0]].evaluateItem + "; " + key[1] + ", " + stat_word[key[1]].evaluateItem + ". <b>Traits:</b> " + t1 + ", " + t2.toLowerCase() + ".</p>";
  b += ev("Origin", life_origin);
  b += ev("Youth", life_youth);
  b += ev("Recent", life_recent);
  b += "<p><b>Arcanum:</b> " + arc + ", unexamined.</p>";
  b += "<p><b>Holding:</b> " + holding_place.evaluateItem + ", " + holding_quirk.evaluateItem + ".</p>";
  b += "<p><b>Look:</b> " + npc_appearance.evaluateItem.toLowerCase() + ". <b>Habit:</b> " + npc_does.evaluateItem.toLowerCase() + ".</p>";
  b += "<p><b>Mark:</b> " + faction_mark.evaluateItem + ".</p>";
  b += "<p><b>Secret:</b> " + npc_secret.evaluateItem.toLowerCase() + ".</p>";
  b += "<p class='close'>Invoke a gift or trait in a dramatic moment for advantage, once per episode, your GM's call. Take disadvantage from a trait instead and tick your arcana ritual. Pick one skill from each life event. Take a specialty in two of them, plus INT extra. Ties are optional: keep the ones you want as threads. Level 1, or average group level. Then roll the first gig.</p>";
  return sec("New hero", b);

trap_damage() =>
  let t = window.sejrTier || 1;
  return [trap_damage_low, trap_damage_mid, trap_damage_high][t - 1].evaluateItem;

rival_check() =>
  if (!window.sejrRival) { window.sejrRival = true; return "<b>Rival crew:</b> " + rival_party_box.evaluateItem; }
  return standard_npc_fallback.evaluateItem;
